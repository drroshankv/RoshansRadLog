Configuration & Sync

Cases are automatically transmitted to your Google Sheet via POST `no-cors`.

Prefills the "Reported By" field so you don't need to retype it.

Google Apps Script Starter Code

Create a Google Sheet, click Extensions → Apps Script, paste this code, and deploy as Web App (Execute as: Me, Who has access: Anyone):

function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); if (sheet.getLastRow() === 0) { sheet.appendRow([ "Timestamp", "Date", "MRN", "Patient", "Modality", "Study", "System", "Organ", "Diagnosis", "Differentials", "Finding", "Status", "ReportedBy", "CheckedBy", "Confirmation" ]); } var data = JSON.parse(e.postData.contents); sheet.appendRow([ new Date(), data.date || "", data.mrn || "", data.patientName || "", data.modality || "", data.studyDescription || "", data.organSystem || "", data.organOfInterest || "", data.primaryDiagnosis || "", data.differentialDiagnoses || "", data.finding || "", data.status || "", data.reportedBy || "", data.checkedBy || "", data.confirmationFollowup || "" ]); return ContentService.createTextOutput("OK"); }
